# Nexum CLI 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增与 UI 同层的 `@nexum/cli` 终端 Surface，复用 `@nexum/surface-adapter` 协议契约，覆盖 REPL 聊天、思考动画、Permission/Question 问答与 model/role-prompt/skill/mcp 管理面命令。

**Architecture:** 先把 `ChatProtocolAdapter` 契约从 `@nexum/chat-surface` 平移到新包 `@nexum/surface-adapter`（chat-surface 改 re-export 壳，公开 API 不破坏）；CLI 聊天链路走适配器，管理面命令直接消费 `@nexum/client`；终端渲染为零依赖的 ANSI 实现（语义内容 ANSI 16 色，装饰元素走 truecolor → 256 → 16 → 静态降级链）。

**Tech Stack:** Bun、TypeScript、`node:util` parseArgs、`node:readline`、ANSI 转义；测试用 `bun test`，注入内存 IO 与脚本化假适配器，不 mock 全局对象。

**设计依据:** `docs/superpowers/specs/2026-11-24-nexum-cli-design.md`；视觉/动画基准 `artifacts/nexum-cli-demo.png`、`nexum-cli-thinking.gif`、`nexum-cli-cool.gif`（双主题）。

/** 终端能力检测与 ANSI 输出助手:语义内容恒为 16 色,装饰元素按能力降级。 */

export type ColorLevel = "none" | "ansi16" | "ansi256" | "truecolor"

export type TermCaps = {
  readonly isTTY: boolean
  readonly color: ColorLevel
  readonly anim: boolean
}

export type CapsOptions = {
  readonly env?: Record<string, string | undefined>
  readonly isTTY?: boolean
  readonly noAnim?: boolean
}

export function detectCaps(options: CapsOptions = {}): TermCaps {
  const env = options.env ?? process.env
  const isTTY = options.isTTY ?? process.stdout.isTTY ?? false
  if (env.NO_COLOR !== undefined || env.TERM === "dumb" || !isTTY) return { isTTY, color: "none", anim: false }
  const color: ColorLevel =
    env.COLORTERM === "truecolor" || env.COLORTERM === "24bit"
      ? "truecolor"
      : env.TERM?.includes("256color")
        ? "ansi256"
        : "ansi16"
  return { isTTY, color, anim: !options.noAnim && env.NEXUM_NO_ANIM === undefined }
}

export type SemanticColor = "green" | "yellow" | "cyan" | "red" | "magenta" | "blue" | "dim" | "bold"

const CODES: Record<SemanticColor, number> = {
  green: 32,
  yellow: 33,
  cyan: 36,
  red: 31,
  magenta: 35,
  blue: 34,
  dim: 2,
  bold: 1,
}

export function style(caps: TermCaps, colors: readonly SemanticColor[], text: string): string {
  if (caps.color === "none" || colors.length === 0) return text
  return `\x1b[${colors.map((color) => CODES[color]).join(";")}m${text}\x1b[0m`
}

export function paint(caps: TermCaps, color: SemanticColor, text: string): string {
  return style(caps, [color], text)
}

type Rgb = readonly [number, number, number]

function hexToRgb(hex: string): Rgb {
  const value = Number.parseInt(hex.slice(1), 16)
  return [(value >> 16) & 255, (value >> 8) & 255, value & 255]
}

function sample(stops: readonly Rgb[], t: number): Rgb {
  const scaled = t * (stops.length - 1)
  const index = Math.min(Math.floor(scaled), stops.length - 2)
  const local = scaled - index
  const [a, b] = [stops[index]!, stops[index + 1]!]
  return [
    Math.round(a[0] + (b[0] - a[0]) * local),
    Math.round(a[1] + (b[1] - a[1]) * local),
    Math.round(a[2] + (b[2] - a[2]) * local),
  ]
}

function to256([r, g, b]: Rgb): number {
  const cube = (v: number) => Math.round((v / 255) * 5)
  return 16 + 36 * cube(r) + 6 * cube(g) + cube(b)
}

/** 装饰性渐变:按能力降级 truecolor → 256 → 16(单色 cyan) → 无。 */
export function gradient(caps: TermCaps, text: string, stops: readonly [string, string, string]): string {
  if (caps.color === "none") return text
  if (caps.color === "ansi16") return paint(caps, "cyan", text)
  const rgb = stops.map(hexToRgb)
  const chars = [...text]
  const out = chars.map((ch, i) => {
    const [r, g, b] = sample(rgb, chars.length <= 1 ? 0 : i / (chars.length - 1))
    return caps.color === "truecolor" ? `\x1b[38;2;${r};${g};${b}m${ch}` : `\x1b[38;5;${to256([r, g, b])}m${ch}`
  })
  return out.join("") + "\x1b[0m"
}
```

/** 动画帧生成器:全部为纯函数,输入 tick 输出字符串,便于测试。 */

import { gradient, paint, style, type TermCaps } from "./term"

export const SPINNER_FRAMES = ["⠋", "⠙", "⠹", "⠸", "⠼", "⠴", "⠦", "⠧", "⠇", "⠏"] as const
export const SPINNER_INTERVAL_MS = 80

export function spinnerFrame(tick: number): string {
  return SPINNER_FRAMES[((tick % SPINNER_FRAMES.length) + SPINNER_FRAMES.length) % SPINNER_FRAMES.length]!
}

/** 呼吸:80ms tick 下 15 tick ≈ 1.2s 周期,前半 dim 后半正常。 */
export function breathing(caps: TermCaps, text: string, tick: number): string {
  return tick % 15 < 8 ? paint(caps, "dim", text) : text
}

/** 扫描光:高亮头 + 5 字符拖尾从左向右扫过,1 步/tick;无色终端原样返回。 */
export function shimmer(caps: TermCaps, text: string, tick: number): string {
  if (caps.color === "none") return text
  const chars = [...text]
  const pos = tick % (chars.length + 5)
  return chars
    .map((ch, i) => {
      const d = pos - i
      if (d === 0) return style(caps, ["cyan", "bold"], ch)
      if (d > 0 && d < 5) return paint(caps, "cyan", ch)
      return paint(caps, "dim", ch)
    })
    .join("")
}

/** 确定进度条:▓ 已完成 + ░ 剩余。 */
export function progressBar(percent: number, width = 13): string {
  const clamped = Math.max(0, Math.min(100, Math.round(percent)))
  const filled = Math.round((clamped / 100) * width)
  return "▓".repeat(filled) + "░".repeat(width - filled)
}

/** 不确定进度条(理发店杆):条纹随 tick 平移,用于无百分比的长任务。 */
export function barberPole(tick: number, width = 13, period = 4): string {
  const unit = "▓▓▓░"
  const offset = tick % period
  return (unit.repeat(Math.ceil((width + period) / unit.length)) + unit).slice(offset, offset + width)
}

export const NEXUM_LOGO = [
  "███╗   ██╗███████╗██╗  ██╗██╗   ██╗███╗   ███╗",
  "████╗  ██║██╔════╝╚██╗██╔╝██║   ██║████╗ ████║",
  "██╔██╗ ██║█████╗   ╚███╔╝ ██║   ██║██╔████╔██║",
  "██║╚██╗██║██╔══╝   ██╔██╗ ██║   ██║██║╚██╔╝██║",
  "██║ ╚████║███████╗██╔╝ ██╗╚██████╔╝██║ ╚═╝ ██║",
  "╚═╝  ╚═══╝╚══════╝╚═╝  ╚═╝ ╚═════╝ ╚═╝     ╚═╝",
] as const

export function logoLines(caps: TermCaps): string[] {
  return NEXUM_LOGO.map((row) => gradient(caps, row, ["#7dcfff", "#bb9af7", "#f7768e"]))
}

/** 输出抽象:生产写 stdout,测试用内存 IO;LiveLine 负责原地刷新的活动行。 */

import { detectCaps, type TermCaps } from "./term"
import { SPINNER_INTERVAL_MS } from "./anim"

export type CliIO = {
  readonly caps: TermCaps
  write(text: string): void
  line(text?: string): void
}

export function createIO(options: { caps?: TermCaps; out?: { write(text: string): unknown } } = {}): CliIO {
  const caps = options.caps ?? detectCaps()
  const out = options.out ?? process.stdout
  return {
    caps,
    write: (text) => void out.write(text),
    line: (text = "") => void out.write(text + "\n"),
  }
}

export function memoryIO(caps: Partial<TermCaps> = {}) {
  let buffer = ""
  return {
    caps: { isTTY: true, color: "none", anim: false, ...caps } as TermCaps,
    write: (text: string) => {
      buffer += text
    },
    line: (text = "") => {
      buffer += text + "\n"
    },
    output: () => buffer,
    plain: () => buffer.replace(/\x1b\[[0-9;]*m/g, ""),
  }
}

export type LiveLine = {
  /** 结束活动行并打印最终内容(静态模式下直接打印)。 */
  settle(text: string): void
  /** 丢弃活动行,不留下任何输出。 */
  cancel(): void
}

/**
 * 活动行:动画开启时按 80ms 原地重写 render(tick),同一时刻只允许一个;
 * 静态模式(非 TTY/无色/--no-anim)不产生任何中间帧。
 */
export function createLiveLine(io: CliIO, render: (tick: number) => string): LiveLine {
  if (!io.caps.anim) {
    return { settle: (text) => io.line(text), cancel: () => {} }
  }
  let tick = 0
  const draw = () => io.write(`\r\x1b[2K${render(tick)}`)
  const timer = setInterval(() => {
    tick += 1
    draw()
  }, SPINNER_INTERVAL_MS)
  draw()
  const stop = () => {
    clearInterval(timer)
    io.write("\r\x1b[2K")
  }
  return {
    settle: (text) => {
      stop()
      io.line(text)
    },
    cancel: stop,
  }
}

/** NexumEvent → 终端文本投影;同一时刻最多一个活动行,终态只认 run.settled 与 session.*。 */

import type { NexumEvent, NexumSessionMessages } from "@nexum/client"
import { barberPole, breathing, shimmer, spinnerFrame } from "./anim"
import { createLiveLine, type CliIO, type LiveLine } from "./io"
import { paint, style, type TermCaps } from "./term"

export type ThinkingMode = "summary" | "full"
export type RunResult = "completed" | "failed" | "cancelled"
export type RenderStatus = "idle" | "running" | "retrying" | "awaiting permission" | "awaiting question" | RunResult

export type Renderer = {
  event(event: NexumEvent): void
  status(): RenderStatus
  thinkingMode(): ThinkingMode
  setThinkingMode(mode: ThinkingMode): void
  expandLastReasoning(): void
  close(): void
}

export function createRenderer(options: { io: CliIO; thinking?: ThinkingMode }): Renderer {
  const io = options.io
  const caps = io.caps
  let mode = options.thinking ?? "summary"
  let status: RenderStatus = "idle"
  let live: LiveLine | undefined
  let liveKind: "thinking" | "tool" | "compaction" | undefined
  let liveToolName = ""
  let liveStartedAt = 0
  let runStartedAt = 0
  let reasoningChars = 0
  let reasoningStartedAt = 0
  let reasoningTail = ""
  let lastReasoning = ""
  let reasoningHeaderPrinted = false
  let assistantHeaderPrinted = false
  let cursorOut = false

  const data = (event: NexumEvent) => (event.data ?? {}) as Record<string, unknown>
  const str = (record: Record<string, unknown>, key: string) => (typeof record[key] === "string" ? (record[key] as string) : "")

  function eraseCursor() {
    if (!cursorOut) return
    io.write("\x1b[D\x1b[K")
    cursorOut = false
  }

  function streamChunk(content: string, dimmed: boolean) {
    eraseCursor()
    io.write(dimmed ? paint(caps, "dim", content) : content)
    io.write(paint(caps, "dim", "▍"))
    cursorOut = true
  }

  function markRunStarted() {
    if (runStartedAt === 0) runStartedAt = Date.now()
  }

  /** 思考块收尾:summary 收敛活动行,full 补打总结行;lastReasoning 保留给 /expand。 */
  function closeReasoning() {
    if (reasoningStartedAt === 0) return
    const elapsed = formatElapsed(Date.now() - reasoningStartedAt)
    const summary = `${paint(caps, "green", "✓")} ${paint(caps, "dim", `思考过程 · ${reasoningChars} 字 · 用时 ${elapsed}`)}`
    if (liveKind === "thinking" && live) {
      live.settle(`${summary}${paint(caps, "dim", "(默认折叠,/expand 查看)")}`)
      live = undefined
      liveKind = undefined
    } else if (mode === "full") {
      eraseCursor()
      io.line(summary)
    }
    reasoningChars = 0
    reasoningStartedAt = 0
    reasoningTail = ""
  }

  function settleLive(text: string) {
    if (!live) return
    live.settle(text)
    live = undefined
    liveKind = undefined
  }

  function cancelLive() {
    if (!live) return
    live.cancel()
    live = undefined
    liveKind = undefined
  }

  function ensureThinkingLive() {
    if (mode !== "summary" || liveKind === "thinking") return
    liveKind = "thinking"
    live = createLiveLine(io, (tick) => {
      const elapsed = ((Date.now() - reasoningStartedAt) / 1000).toFixed(1)
      const label = caps.color === "truecolor" || caps.color === "ansi256" ? shimmer(caps, "正在思考", tick) : breathing(caps, "正在思考", tick)
      const tail = reasoningTail ? paint(caps, "dim", ` · ${reasoningTail}`) : ""
      return `${paint(caps, "cyan", spinnerFrame(tick))} ${label} · 已 ${reasoningChars} 字 · ${elapsed}s${tail}`
    })
  }

  function printAgentHeader() {
    if (assistantHeaderPrinted) return
    assistantHeaderPrinted = true
    io.line(style(caps, ["magenta", "bold"], "Agent"))
  }

  function event(event: NexumEvent): void {
    const record = data(event)
    switch (event.type) {
      case "assistant.thinking":
        status = "running"
        markRunStarted()
        if (reasoningStartedAt === 0) reasoningStartedAt = Date.now()
        ensureThinkingLive()
        return
      case "assistant.reasoning": {
        status = "running"
        markRunStarted()
        if (reasoningStartedAt === 0) reasoningStartedAt = Date.now()
        const content = str(record, "content")
        reasoningChars += [...content].length
        lastReasoning += content
        reasoningTail = [...lastReasoning].slice(-30).join("")
        if (mode === "full") {
          if (!reasoningHeaderPrinted) {
            reasoningHeaderPrinted = true
            io.line(paint(caps, "dim", "思考全文:"))
          }
          streamChunk(content, true)
        } else {
          ensureThinkingLive()
        }
        return
      }
      case "assistant.text": {
        status = "running"
        markRunStarted()
        closeReasoning()
        printAgentHeader()
        streamChunk(str(record, "content"), false)
        return
      }
      case "tool.request": {
        status = "running"
        markRunStarted()
        closeReasoning()
        eraseCursor()
        cancelLive()
        liveToolName = str(record, "toolName") || str(record, "toolCallID") || "工具"
        liveKind = "tool"
        liveStartedAt = Date.now()
        const name = liveToolName
        live = createLiveLine(io, (tick) => `${paint(caps, "cyan", spinnerFrame(tick))} ${paint(caps, "dim", `工具调用 ${name} · ${((Date.now() - liveStartedAt) / 1000).toFixed(1)}s`)}`)
        return
      }
      case "tool.result": {
        status = "running"
        const ok = record.ok !== false
        const name = str(record, "toolName") || liveToolName || "工具"
        const summary = compact(str(record, "output"))
        const mark = ok ? paint(caps, "green", "✓") : paint(caps, "red", "✗")
        settleLive(`${mark} ${paint(caps, "dim", `工具结果 ${name}${summary ? ` · ${summary}` : ""}`)}`)
        liveToolName = ""
        return
      }
      case "permission.requested":
        status = "awaiting permission"
        closeReasoning()
        eraseCursor()
        return
      case "question.request":
        status = "awaiting question"
        closeReasoning()
        eraseCursor()
        return
      case "question.result":
      case "question.cancelled":
        status = "running"
        return
      case "context.compaction.started":
        status = "running"
        cancelLive()
        liveKind = "compaction"
        live = createLiveLine(io, (tick) => `${paint(caps, "cyan", spinnerFrame(tick))} ${paint(caps, "dim", "上下文压缩")} ${paint(caps, "cyan", barberPole(tick))}`)
        return
      case "context.compaction.completed":
        settleLive(`${paint(caps, "green", "✓")} ${paint(caps, "dim", "上下文压缩完成")}`)
        return
      case "context.compaction.failed":
        settleLive(`${paint(caps, "red", "✗")} ${paint(caps, "dim", "上下文压缩失败")}`)
        return
      case "context.project.loaded":
        io.line(paint(caps, "dim", `项目上下文已加载 ${Array.isArray(record.sources) ? record.sources.length : 0} 个来源`))
        return
      case "skill.activated":
        io.line(`${paint(caps, "green", "✓")} Skill ${paint(caps, "cyan", `@${str(record, "skillID")}`)} 已进入上下文`)
        return
      case "skill.failed":
        io.line(`${paint(caps, "red", "✗")} Skill 激活失败 ${paint(caps, "dim", str(record, "code"))}`)
        return
      case "model.attempt":
        status = "running"
        markRunStarted()
        io.line(paint(caps, "dim", `模型尝试 · 第 ${String(record.attempt ?? "?")} 次调用`))
        return
      case "model.retry":
        status = "retrying"
        io.line(paint(caps, "yellow", `模型重试 · ${String(record.delayMs ?? 0)}ms 后重试`))
        return
      case "coding-loop.limit.reached":
        io.line(paint(caps, "yellow", "已达到本轮步数上限,/continue 可继续"))
        return
      case "run.settled": {
        const result = record.result === "failed" || record.result === "cancelled" ? record.result : "completed"
        status = result
        closeReasoning()
        eraseCursor()
        cancelLive()
        const elapsed = runStartedAt === 0 ? "" : ` · 用时 ${formatElapsed(Date.now() - runStartedAt)}`
        if (result === "completed") io.line(`${paint(caps, "green", "✓ 运行完成")}${paint(caps, "dim", elapsed)}`)
        if (result === "failed") io.line(paint(caps, "red", "✗ 运行失败"))
        if (result === "cancelled") io.line(paint(caps, "dim", "— 已取消运行"))
        runStartedAt = 0
        reasoningChars = 0
        reasoningStartedAt = 0
        lastReasoning = ""
        reasoningHeaderPrinted = false
        assistantHeaderPrinted = false
        return
      }
      case "session.completed":
        status = "completed"
        return
      case "session.failed":
        status = "failed"
        io.line(paint(caps, "red", `运行失败 ${str(record, "message")}`))
        return
      case "session.cancelled":
        status = "cancelled"
        return
      default:
        return
    }
  }

  return {
    event,
    status: () => status,
    thinkingMode: () => mode,
    setThinkingMode: (next) => {
      mode = next
    },
    expandLastReasoning: () => {
      if (!lastReasoning) {
        io.line(paint(caps, "dim", "暂无可展开的思考过程"))
        return
      }
      eraseCursor()
      io.line(paint(caps, "dim", "思考全文"))
      io.line(paint(caps, "dim", lastReasoning))
    },
    close: () => {
      closeReasoning()
      eraseCursor()
      cancelLive()
    },
  }
}

function formatElapsed(ms: number): string {
  if (ms < 1000) return `${Math.max(1, Math.round(ms))}ms`
  return `${(ms / 1000).toFixed(1)}s`
}

function compact(value: string): string {
  return value.replace(/\s+/g, " ").trim().slice(0, 60)
}

/** 历史消息 → 紧凑 transcript 行。 */
export function historyLines(caps: TermCaps, messages: NexumSessionMessages["messages"]): string[] {
  const lines: string[] = []
  for (const entry of messages) {
    const message = entry.message
    if (message.role === "user") {
      lines.push(`${paint(caps, "green", "你")} ${paint(caps, "dim", "历史")}`)
      lines.push(`  ${message.content}`)
      continue
    }
    if (message.role === "assistant") {
      if (message.reasoningContent !== undefined) lines.push(paint(caps, "dim", `  思考过程 · ${[...message.reasoningContent].length} 字(历史)`))
      if (message.content !== undefined) {
        lines.push(`${style(caps, ["magenta", "bold"], "Agent")} ${paint(caps, "dim", "历史")}`)
        lines.push(`  ${message.content}`)
      }
      for (const toolCall of message.toolCalls ?? []) lines.push(paint(caps, "dim", `  ● 工具调用 ${toolCall.name}(历史)`))
      continue
    }
    if (message.role === "tool") {
      lines.push(paint(caps, "dim", `  ✓ 工具结果 · ${compact(message.content)}`))
      continue
    }
    lines.push(paint(caps, "dim", `  系统消息 · ${message.sourceID ?? "context"}`))
  }
  return lines
}

/** Permission / Question 的终端问答流:编号选项,按题序提交二维数组。 */

import type { NexumPermission, NexumQuestion } from "@nexum/client"
import type { CliIO } from "./io"
import { paint, style } from "./term"

export type Prompter = (query: string) => Promise<string>

export type PermissionDecision = "allow_once" | "allow_always" | "deny"

export async function askPermission(io: CliIO, prompter: Prompter, request: NexumPermission): Promise<PermissionDecision> {
  io.line(style(io.caps, ["yellow", "bold"], "⚠ 需要权限确认"))
  io.line(`  ${style(io.caps, ["bold"], request.action)} · ${request.resource}`)
  if (request.reason) io.line(paint(io.caps, "dim", `  原因: ${request.reason}`))
  io.line(`  ${paint(io.caps, "blue", "[1]")} 允许一次   ${paint(io.caps, "blue", "[2]")} 始终允许   ${paint(io.caps, "blue", "[3]")} 拒绝`)
  while (true) {
    const answer = (await prompter(paint(io.caps, "cyan", "选择 › "))).trim()
    if (answer === "1") return "allow_once"
    if (answer === "2") return "allow_always"
    if (answer === "3") return "deny"
    io.line(paint(io.caps, "dim", "请输入 1、2 或 3"))
  }
}

export async function askQuestion(io: CliIO, prompter: Prompter, request: NexumQuestion): Promise<readonly (readonly string[])[] | "cancel"> {
  io.line(style(io.caps, ["cyan", "bold"], "? 需要补充信息"))
  const answers: string[][] = []
  for (const [index, question] of request.questions.entries()) {
    io.line(`  ${style(io.caps, ["bold"], `${index + 1}. ${question.question}`)}`)
    question.options.forEach((option, optionIndex) => {
      const description = option.description ? paint(io.caps, "dim", ` — ${option.description}`) : ""
      io.line(`     ${paint(io.caps, "blue", `${optionIndex + 1})`)} ${option.label}${description}`)
    })
    while (true) {
      const hint = question.multiple ? "回答(多选逗号分隔,c 取消) › " : "回答(c 取消) › "
      const answer = (await prompter(paint(io.caps, "cyan", hint))).trim()
      if (answer === "c") return "cancel"
      const picks = answer.split(/[,\s]+/).filter(Boolean)
      const indexes = picks.map((pick) => Number.parseInt(pick, 10) - 1)
      const valid =
        picks.length > 0 &&
        indexes.every((i) => Number.isInteger(i) && i >= 0 && i < question.options.length) &&
        (question.multiple === true || indexes.length === 1)
      if (!valid) {
        io.line(paint(io.caps, "dim", "无效选择,请重试"))
        continue
      }
      answers.push(indexes.map((i) => question.options[i]!.label))
      break
    }
  }
  return answers
}

export function permissionFromEvent(data: Record<string, unknown>, sessionID: string): NexumPermission | undefined {
  const id = stringValue(data.requestID)
  const action = stringValue(data.action)
  const resource = stringValue(data.resource)
  if (!id || !action || !resource) return undefined
  const reason = stringValue(data.reason)
  return { id, sessionID, action, resource, ...(reason ? { reason } : {}) }
}

export function questionFromEvent(data: Record<string, unknown>, sessionID: string): NexumQuestion | undefined {
  const id = stringValue(data.requestID)
  if (!id || !Array.isArray(data.questions)) return undefined
  const questions = data.questions.filter(isQuestion)
  if (questions.length === 0) return undefined
  return { id, sessionID, toolCallID: stringValue(data.toolCallID), questions }
}

function isQuestion(value: unknown): value is NexumQuestion["questions"][number] {
  if (!value || typeof value !== "object") return false
  const question = value as Record<string, unknown>
  return typeof question.header === "string" && typeof question.question === "string" && Array.isArray(question.options) && question.options.every(isOption)
}

function isOption(value: unknown): value is { label: string; description: string } {
  if (!value || typeof value !== "object") return false
  const option = value as Record<string, unknown>
  return typeof option.label === "string" && typeof option.description === "string"
}

function stringValue(value: unknown): string {
  return typeof value === "string" ? value : ""
}

/** REPL 交互循环:历史回放 → 事件订阅 → 输入分发,交互请求走顺序队列。 */

import { createInterface } from "node:readline/promises"
import type { NexumEvent } from "@nexum/client"
import type { ChatProtocolAdapter } from "@nexum/surface-adapter"
import { askPermission, askQuestion, permissionFromEvent, questionFromEvent, type Prompter } from "./interact"
import { createRenderer, historyLines } from "./render"
import { paint } from "./term"
import type { CliIO } from "./io"

export type LineSource = {
  next(): Promise<string | null>
  close(): void
}

export function arrayLineSource(lines: readonly string[]): LineSource {
  let index = 0
  return {
    next: async () => (index < lines.length ? lines[index++]! : null),
    close: () => {},
  }
}

export function readlineLineSource(skills: () => readonly string[]): LineSource & { prompter: Prompter } {
  const rl = createInterface({
    input: process.stdin,
    output: process.stdout,
    completer: (line: string) => {
      const match = line.match(/(?:^|\s)@([a-z0-9-]*)$/i)
      if (!match) return [[], line]
      const query = match[1]?.toLowerCase() ?? ""
      const hits = skills()
        .filter((id) => id.toLowerCase().startsWith(query))
        .map((id) => `@${id} `)
      return [hits, line.slice(0, line.length - match[1]!.length - 1)]
    },
  })
  return {
    next: () => rl.question("nexum ▸ ").then((answer) => answer, () => null),
    prompter: (query) => rl.question(query),
    close: () => rl.close(),
  }
}

const HELP = `可用命令:
  /help        显示帮助          /new [名称]   新建会话
  /sessions    列出会话          /use <id>     切换会话
  /skills      列出可用 Skill    /events       最近事件
  /cancel      取消当前运行      /continue     继续上次运行
  /thinking    summary|full      /expand       展开最近思考
  /quit        退出`

export type ReplOptions = {
  readonly adapter: ChatProtocolAdapter
  readonly io: CliIO
  readonly sessionID: string
  readonly lineSource: LineSource
  readonly prompter: Prompter
}

export async function runRepl(options: ReplOptions): Promise<void> {
  const { adapter, io } = options
  const sessionID = options.sessionID
  const renderer = createRenderer({ io })
  const recentEvents: string[] = []

  async function openSession(id: string) {
    const [state, history] = await Promise.all([adapter.loadState(id), adapter.loadHistory({ sessionID: id, limit: 50 })])
    io.line(paint(io.caps, "dim", `会话 ${id} · ${state.session.name} · 已回放历史 ${history.messages.length} 条`))
    for (const line of historyLines(io.caps, history.messages)) io.line(line)
    return state.lastSequence
  }

  let afterSequence = await openSession(sessionID)

  const controller = new AbortController()
  const seen = new Set<string>()
  let interactionQueue = Promise.resolve()

  function enqueueInteraction(event: NexumEvent) {
    interactionQueue = interactionQueue.then(async () => {
      const data = (event.data ?? {}) as Record<string, unknown>
      if (event.type === "permission.requested") {
        const request = permissionFromEvent(data, sessionID)
        if (!request) return
        const decision = await askPermission(io, options.prompter, request)
        await adapter.replyPermission({ sessionID, requestID: request.id, decision })
        io.line(paint(io.caps, "green", decision === "deny" ? "权限已拒绝" : "权限已通过"))
        return
      }
      const request = questionFromEvent(data, sessionID)
      if (!request) return
      const answers = await askQuestion(io, options.prompter, request)
      if (answers === "cancel") {
        await adapter.cancelQuestion({ sessionID, requestID: request.id })
        io.line(paint(io.caps, "dim", "已取消补充信息"))
        return
      }
      await adapter.replyQuestion({ sessionID, requestID: request.id, answers })
      io.line(paint(io.caps, "green", "补充信息已提交,运行继续"))
    })
  }

  const pump = (async () => {
    try {
      for await (const event of adapter.subscribeEvents({ sessionID, afterSequence, signal: controller.signal })) {
        if (controller.signal.aborted) return
        if (seen.has(event.eventID)) continue
        seen.add(event.eventID)
        afterSequence = event.sequence
        recentEvents.push(`${event.sequence} ${event.type}`)
        if (recentEvents.length > 50) recentEvents.shift()
        renderer.event(event)
        if (event.type === "permission.requested" || event.type === "question.request") enqueueInteraction(event)
      }
    } catch {
      if (!controller.signal.aborted) io.line(paint(io.caps, "yellow", "事件流已断开,重新输入任意内容将继续"))
    }
  })()

  const locked = () => ["running", "retrying", "awaiting permission", "awaiting question"].includes(renderer.status())

  try {
    while (true) {
      const line = await options.lineSource.next()
      if (line === null) break
      const text = line.trim()
      if (!text) continue
      if (text.startsWith("/")) {
        const [command, ...rest] = text.slice(1).split(/\s+/)
        if (command === "quit" || command === "exit") break
        if (command === "help") io.line(HELP)
        else if (command === "cancel") {
          await adapter.cancelRun(sessionID)
          io.line(paint(io.caps, "dim", "已请求取消运行"))
        } else if (command === "continue") {
          await adapter.continueRun(sessionID)
          io.line(paint(io.caps, "dim", "正在继续运行"))
        } else if (command === "skills") {
          const skills = await adapter.listSkills()
          const ready = skills.filter((skill) => skill.status === "ready")
          if (ready.length === 0) io.line(paint(io.caps, "dim", "暂无可用 Skill"))
          for (const skill of ready) io.line(`${paint(io.caps, "cyan", `@${skill.id}`)} ${paint(io.caps, "dim", skill.description)}`)
        } else if (command === "events") {
          if (recentEvents.length === 0) io.line(paint(io.caps, "dim", "本轮还没有事件"))
          for (const item of recentEvents.slice(-20)) io.line(paint(io.caps, "dim", item))
        } else if (command === "thinking") {
          const mode = rest[0] === "full" ? "full" : "summary"
          renderer.setThinkingMode(mode)
          io.line(paint(io.caps, "dim", `思考展示:${mode}`))
        } else if (command === "expand") renderer.expandLastReasoning()
        else if (command === "new" || command === "use" || command === "sessions") io.line(paint(io.caps, "yellow", "会话管理请使用一次性命令:nexum session list / nexum --session <id>"))
        else io.line(paint(io.caps, "yellow", `未知命令 /${command},输入 /help 查看帮助`))
        continue
      }
      if (locked()) {
        io.line(paint(io.caps, "yellow", "Agent 正在回复,/cancel 取消本轮"))
        continue
      }
      await adapter.sendMessage({ sessionID, text })
    }
  } finally {
    controller.abort()
    options.lineSource.close()
    await interactionQueue.catch(() => {})
    await pump.catch(() => {})
    renderer.close()
  }
}

import { Nexum } from "@nexum/client"
import type { ChatProtocolAdapter } from "@nexum/surface-adapter"
import { askPermission, askQuestion, permissionFromEvent, questionFromEvent, type Prompter } from "./interact"
import { createRenderer } from "./render"
import { paint } from "./term"
import type { CliIO } from "./io"

export type NexumClient = ReturnType<typeof Nexum.make>

const STATUS_LABELS: Record<string, string> = {
  idle: "空闲",
  running: "执行中",
  completed: "已完成",
  failed: "失败",
  cancelled: "已取消",
}

function statusText(io: CliIO, status: string): string {
  const label = STATUS_LABELS[status] ?? status
  if (status === "completed") return paint(io.caps, "green", label)
  if (status === "failed") return paint(io.caps, "red", label)
  if (status === "cancelled") return paint(io.caps, "dim", label)
  if (status === "running") return paint(io.caps, "cyan", label)
  return paint(io.caps, "dim", label)
}

export async function cmdSessionList(client: NexumClient, io: CliIO): Promise<void> {
  const sessions = (await client.experimental.nexum.session.list()).data
  io.line(`${paint(io.caps, "dim", "ID        名称            目录                状态     更新于")}`)
  for (const session of sessions) {
    const id = paint(io.caps, "blue", session.id.slice(0, 8).padEnd(8))
    const name = session.name.slice(0, 14).padEnd(14)
    const directory = session.directory.slice(-18).padEnd(18)
    io.line(`${id}  ${name}  ${directory}  ${statusText(io, session.status)}  ${session.updatedAt.slice(0, 16).replace("T", " ")}`)
  }
  if (sessions.length === 0) io.line(paint(io.caps, "dim", "暂无会话,nexum session create 创建"))
}

export async function cmdSessionCreate(client: NexumClient, io: CliIO, input: { directory: string; name?: string }): Promise<void> {
  const session = (await client.experimental.nexum.session.create(input)).data
  io.line(`${paint(io.caps, "green", "✓")} 已创建会话 ${paint(io.caps, "blue", session.id)}`)

export async function resolveSessionID(client: NexumClient, io: CliIO, requested?: string): Promise<string> {
  if (requested) return requested
  const sessions = (await client.experimental.nexum.session.list()).data
  const latest = [...sessions].sort((a, b) => b.updatedAt.localeCompare(a.updatedAt))[0]
  if (latest) return latest.id
  const created = (await client.experimental.nexum.session.create({ directory: process.cwd() })).data
  io.line(paint(io.caps, "dim", `已创建会话 ${created.id}`))
  return created.id
}

export type PromptOptions = {
  readonly client: NexumClient
  readonly adapter: ChatProtocolAdapter
  readonly io: CliIO
  readonly prompter: Prompter
  readonly text: string
  readonly sessionID?: string
  readonly directory: string
}

export async function cmdPrompt(options: PromptOptions): Promise<number> {
  const { client, adapter, io, prompter } = options
  const sessionID = options.sessionID ?? (await client.experimental.nexum.session.create({ directory: options.directory })).data.id
  if (!options.sessionID) io.line(paint(io.caps, "dim", `会话 ${sessionID}`))
  const renderer = createRenderer({ io })
  let settled: "completed" | "failed" | "cancelled" | undefined
  let interrupted = false
  const onSigint = () => {
    interrupted = true
    void adapter.cancelRun(sessionID)
  }
  process.once("SIGINT", onSigint)
  try {
    await adapter.sendMessage({ sessionID, text: options.text })
    for await (const event of adapter.subscribeEvents({ sessionID })) {
      renderer.event(event)
      if (event.type === "permission.requested") {
        const request = permissionFromEvent((event.data ?? {}) as Record<string, unknown>, sessionID)
        if (!request) continue
        if (!io.caps.isTTY) {
          await adapter.replyPermission({ sessionID, requestID: request.id, decision: "deny" })
          io.line(paint(io.caps, "yellow", "非交互环境,权限已自动拒绝"))
          continue
        }
        const decision = await askPermission(io, prompter, request)
        await adapter.replyPermission({ sessionID, requestID: request.id, decision })
        continue
      }
      if (event.type === "question.request") {
        const request = questionFromEvent((event.data ?? {}) as Record<string, unknown>, sessionID)
        if (!request) continue
        if (!io.caps.isTTY) {
          await adapter.cancelQuestion({ sessionID, requestID: request.id })
          io.line(paint(io.caps, "yellow", "非交互环境,补充信息已取消"))
          continue
        }
        const answers = await askQuestion(io, prompter, request)
        if (answers === "cancel") await adapter.cancelQuestion({ sessionID, requestID: request.id })
        else await adapter.replyQuestion({ sessionID, requestID: request.id, answers })
        continue
      }
      if (event.type === "run.settled") {
        const data = (event.data ?? {}) as Record<string, unknown>
        settled = data.result === "failed" || data.result === "cancelled" ? data.result : "completed"
        break
      }
    }
  } finally {
    process.removeListener("SIGINT", onSigint)
    renderer.close()
  }
  if (settled === "completed") return 0
  if (settled === "cancelled" || interrupted) return 130
  return 1
}

  }

  openBrowser(attempt.authorizationUrl)
  const redirectUrl = (await prompter("粘贴回调 URL › ")).trim()
  if (!redirectUrl) {

    io.line(paint(io.caps, "dim", "已取消授权"))
    return 130
  }
  await client.experimental.nexum.mcp.completeAuthorization({ attemptID: attempt.id, redirectUrl })
  const deadline = Date.now() + 120_000
  while (Date.now() < deadline) {
    const status = (await client.experimental.nexum.mcp.authorizationStatus({ attemptID: attempt.id })).data
    if (status.status === "authorized") {
      io.line(`${paint(io.caps, "green", "✓")} 已连接(authorized)`)
      return 0
    }
    if (status.status === "failed" || status.status === "expired" || status.status === "cancelled") {
      io.line(paint(io.caps, "red", `授权${status.status === "failed" ? "失败" : status.status === "expired" ? "过期" : "已取消"}`))
      return 1
    }
    await new Promise((resolve) => setTimeout(resolve, 1000))
  }
  io.line(paint(io.caps, "red", "授权超时"))
  return 1
}

  io.line(paint(io.caps, "dim", `共 ${prompts.length} 个提示`))
}

function openBrowser(url: string): void {
  const command = process.platform === "darwin" ? "open" : process.platform === "win32" ? "start" : "xdg-open"
  const child = spawn(command, [url], { detached: true, stdio: "ignore" })
  child.on("error", () => {})
  child.unref()
}

#!/usr/bin/env bun
/** Nexum CLI 入口:参数解析、子命令分发、退出码约定(0 成功/1 失败/2 用法错误/130 取消)。 */

import { parseArgs } from "node:util"
import { Nexum, NexumClientError } from "@nexum/client"
import { createChatProtocolAdapter } from "@nexum/surface-adapter"
import {
  cmdMcpAdd,
  cmdMcpConnect,
  cmdMcpDisconnect,
  cmdMcpList,
  cmdMcpPrompts,
  cmdMcpRemove,
  cmdMcpResources,
  cmdMcpTools,
  cmdModelGet,
  cmdModelProbe,
  cmdModelSet,
  cmdRolePromptGet,
  cmdRolePromptReset,
  cmdRolePromptSet,
  cmdSkillDetail,
  cmdSkillList,
  cmdSkillRefresh,
  cmdSkillRegister,
  cmdSkillUnregister,
  cmdSkillValidate,
} from "./admin"
import { cmdPrompt, cmdSessionCreate, cmdSessionList, resolveSessionID } from "./commands"
import { createIO } from "./io"
import { readlineLineSource, runRepl } from "./repl"
import { detectCaps } from "./term"

export const USAGE = `Nexum CLI — 与 Web UI 同级的终端 Surface

用法: nexum [--server <url>] [--session <id>] [--no-anim] [命令]

  (无命令)                进入 REPL(复用最近会话或新建)
  session list            列出会话
  session create [--dir <路径>] [--name <名称>]
  prompt <文本> [--session <id>]   流式执行,退出码 0/1/130
  model get | set | probe
  role-prompt get | set <文件> | reset
  skill list | detail <id> | validate [id] | refresh [id] | register <地址> | unregister <id>
  mcp list | add | connect <id> | disconnect <id> | remove <id> | tools <id> | resources <id> | prompts <id>

选项:
  --server    Server 地址(默认 $NEXUM_API_URL 或 http://127.0.0.1:4097)
  --session   指定会话 ID
  --no-anim   禁用动画(非 TTY 自动禁用)
  --help      显示帮助`

export function resolveBaseUrl(flag: string | undefined, env: Record<string, string | undefined>): string {
  return flag ?? env.NEXUM_API_URL ?? "http://127.0.0.1:4097"
}

export async function run(argv: string[]): Promise<number> {
  const { values, positionals } = parseArgs({
    args: argv,
    allowPositionals: true,
    options: {
      server: { type: "string" },
      session: { type: "string" },
      dir: { type: "string" },
      name: { type: "string" },
      "no-anim": { type: "boolean", default: false },
      help: { type: "boolean", default: false },
    },
  })
  if (values.help) {
    console.log(USAGE)
    return 0
  }
  const io = createIO({ caps: detectCaps({ noAnim: values["no-anim"] }) })
  const baseUrl = resolveBaseUrl(values.server, process.env)
  const client = Nexum.make({ baseUrl })
  try {
    await client.capabilities()
  } catch {
    io.line(`无法连接 ${baseUrl}`)
    io.line("请先启动 Server:bun run --cwd packages/nexum-server dev")
    return 1
  }
  const adapter = createChatProtocolAdapter({ baseUrl })
  const [command, ...rest] = positionals
  try {
    if (command === undefined) {
      const sessionID = await resolveSessionID(client, io, values.session)
      const skills = await adapter.listSkills()
      const lineSource = readlineLineSource(() => skills.filter((skill) => skill.status === "ready").map((skill) => skill.id))
      await runRepl({ adapter, io, sessionID, lineSource, prompter: lineSource.prompter })
      return 0
    }
    const prompterSource = readlineLineSource(() => [])
    try {
      if (command === "session") {
        if (rest[0] === "list") return await cmdSessionList(client, io).then(() => 0)
        if (rest[0] === "create") return await cmdSessionCreate(client, io, { directory: values.dir ?? process.cwd(), ...(values.name ? { name: values.name } : {}) }).then(() => 0)
      }
      if (command === "prompt") {
        const text = rest.join(" ").trim()
        if (!text) {
          io.line("用法:nexum prompt <文本> [--session <id>]")
          return 2
        }
        return await cmdPrompt({ client, adapter, io, prompter: prompterSource.prompter, text, ...(values.session ? { sessionID: values.session } : {}), directory: process.cwd() })
      }
      if (command === "model") {
        if (rest[0] === "get") return await cmdModelGet(client, io).then(() => 0)
        if (rest[0] === "set") return await cmdModelSet(client, io, prompterSource.prompter).then(() => 0)
        if (rest[0] === "probe") return await cmdModelProbe(client, io).then(() => 0)
      }
      if (command === "role-prompt") {
        if (rest[0] === "get") return await cmdRolePromptGet(client, io).then(() => 0)
        if (rest[0] === "set" && rest[1]) return await cmdRolePromptSet(client, io, rest[1]).then(() => 0)
        if (rest[0] === "reset") return await cmdRolePromptReset(client, io).then(() => 0)
      }
      if (command === "skill") {
        if (rest[0] === "list") return await cmdSkillList(client, io).then(() => 0)
        if (rest[0] === "detail" && rest[1]) return await cmdSkillDetail(client, io, rest[1]).then(() => 0)
        if (rest[0] === "validate") return await cmdSkillValidate(client, io, rest[1]).then(() => 0)
        if (rest[0] === "refresh") return await cmdSkillRefresh(client, io, rest[1]).then(() => 0)
        if (rest[0] === "register" && rest[1]) return await cmdSkillRegister(client, io, rest[1]).then(() => 0)
        if (rest[0] === "unregister" && rest[1]) return await cmdSkillUnregister(client, io, rest[1]).then(() => 0)
      }
      if (command === "mcp") {
        if (rest[0] === "list") return await cmdMcpList(client, io).then(() => 0)
        if (rest[0] === "add") return await cmdMcpAdd(client, io, prompterSource.prompter).then(() => 0)
        if (rest[0] === "connect" && rest[1]) return await cmdMcpConnect(client, io, prompterSource.prompter, rest[1])
        if (rest[0] === "disconnect" && rest[1]) return await cmdMcpDisconnect(client, io, rest[1]).then(() => 0)
        if (rest[0] === "remove" && rest[1]) return await cmdMcpRemove(client, io, rest[1]).then(() => 0)
        if (rest[0] === "tools" && rest[1]) return await cmdMcpTools(client, io, rest[1]).then(() => 0)
        if (rest[0] === "resources" && rest[1]) return await cmdMcpResources(client, io, rest[1]).then(() => 0)
        if (rest[0] === "prompts" && rest[1]) return await cmdMcpPrompts(client, io, rest[1]).then(() => 0)
      }
      io.line(`未知命令:${positionals.join(" ")}\n`)
      io.line(USAGE)
      return 2
    } finally {
      prompterSource.close()
    }
  } catch (error) {
    if (error instanceof NexumClientError) {
      io.line(`${error.code}: ${error.message}${error.traceId ? `(traceId: ${error.traceId})` : ""}`)
      return 1
    }
    io.line(error instanceof Error ? error.message : "命令执行失败")
    return 1
  }
}

if (import.meta.main) {
  const code = await run(process.argv.slice(2))
  process.exit(code)
}
```
